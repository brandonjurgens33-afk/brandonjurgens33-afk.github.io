records_regular <- logs |>
  filter(season_type ==2)
regular_wins <- records_regular |>
  group_by(team_short_display_name) |>
  summarize(
    wins = sum(team_winner == TRUE),
               losses = sum(team_winner == FALSE)
  )
auto_bids <- c(
  "Liberty", "Winthrop", "Hofstra", "Northern Kentucky", "Yale", "Bradley", "Utah St", "Robert Morris", "Belmont", "Boston Univ", "ETSU", "N Dakota St", "Gonzaga", "Florida St", "Vermont", "Cincinnati", "Dayton", "Kansas", "Creighton", "E Washington", "Wisconsin", "UC Irvine", "North Texas", "Siena", "Akron", "NC Central", "Oregon", "Kentucky", "SF Austin", "Prairie View", "Little Rock", "New Mexico St"
)
automatic <- regular_wins |>
  filter(team_short_display_name %in% auto_bids) |>
  mutate (selection = "Conference Champions")
at_large <- regular_wins |>
  filter(team_short_display_name %in% auto_bids) |>
  arrange(desc(wins)) |>
  slice_head(n= 68 - nrow(automatic)) |>
  mutate(selection = "At Large")
field_68 <- bind_rows(automatic, at_large)
